
# --- Waveshare 1.44" LCD HAT (ST7735S 128x128) ---
# Appended to /boot/firmware/config.txt by install.sh.
# Display: kernel panel-mipi-dbi driver -> /dev/fb0 + DRM, init commands in
# /lib/firmware/waveshare144.bin (firmware/mkpanel.py).
[all]
dtparam=spi=on
dtoverlay=mipi-dbi-spi,spi0-0,speed=32000000
dtparam=compatible=waveshare144\0panel-mipi-dbi-spi
dtparam=width=128,height=128,width-mm=26,height-mm=26
dtparam=x-offset=1,y-offset=2
dtparam=reset-gpio=27,dc-gpio=25,backlight-gpio=24
dtparam=write-only
# HAT buttons -> keyboard keys (gpio_pull=2 = pull-up)
dtoverlay=gpio-key,gpio=6,active_low=1,gpio_pull=2,keycode=103,label=UP
dtoverlay=gpio-key,gpio=19,active_low=1,gpio_pull=2,keycode=108,label=DOWN
dtoverlay=gpio-key,gpio=5,active_low=1,gpio_pull=2,keycode=105,label=LEFT
dtoverlay=gpio-key,gpio=26,active_low=1,gpio_pull=2,keycode=106,label=RIGHT
dtoverlay=gpio-key,gpio=13,active_low=1,gpio_pull=2,keycode=28,label=PRESS
dtoverlay=gpio-key,gpio=21,active_low=1,gpio_pull=2,keycode=2,label=KEY1
dtoverlay=gpio-key,gpio=20,active_low=1,gpio_pull=2,keycode=3,label=KEY2
dtoverlay=gpio-key,gpio=16,active_low=1,gpio_pull=2,keycode=4,label=KEY3
# --- end Waveshare 1.44" LCD HAT ---
