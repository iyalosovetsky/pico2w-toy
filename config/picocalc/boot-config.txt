
# --- PicoCalc + Raspberry Pi Zero 2 W ---
# Reference: what /boot/firmware/config.txt needs. The display and keyboard
# lines come from https://github.com/ironat/picocalc_trixie (Steps 4-5: it also
# installs /lib/firmware/picomipi.bin and builds the picocalc_kbd driver);
# install.sh only adds the audio lines when they are missing.
dtparam=i2c_arm=on
dtparam=spi=on
dtoverlay=picocalc_kbd
# Display: 320x320 ST7365P via the kernel panel-mipi-dbi driver -> /dev/fb0
dtoverlay=mipi-dbi-spi,spi0-0,speed=70000000
dtparam=compatible=picomipi\0panel-mipi-dbi-spi
dtparam=width=320,height=320,width-mm=43,height-mm=43
dtparam=reset-gpio=25,dc-gpio=24
dtparam=backlight-gpio=18
dtparam=clock-frequency=50
# Sound: PWM on GPIO 12/13 -> the PicoCalc's amplifier and speakers
dtparam=audio=on
dtoverlay=audremap,pins_12_13
# UART on GPIO 14/15 (/dev/serial0 = ttyS0) to the ESP32-C6-Zero; Bluetooth keeps the PL011.
# install.sh also takes console=serial0 out of cmdline.txt so programs can use the port.
enable_uart=1
# --- end PicoCalc ---
